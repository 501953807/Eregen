# Startup Script System Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Create a unified shell-based startup script system for all Eregen subsystems with configurable ports, dependency checking, and cross-platform compatibility.

**Architecture:** Root aggregator script (`scripts/start.sh`) delegates to shared libraries in `scripts/lib/`. Each subsystem gets lightweight wrapper scripts. Port config lives in root `.env`. Zero runtime dependencies beyond bash.

**Tech Stack:** POSIX-compatible Shell (bash/zsh), no external tools required for core logic.

## Global Constraints

- Non-Docker native development is the primary mode; Docker compose is alternative via `--docker` flag
- Ports are configurable via root `.env` file with CLI override support
- Each subsystem has its own `start.sh`/`stop.sh` that works independently of the root aggregator
- Firmware scripts handle compilation only (no port management)
- Clean scope: only removes PID files, lock files, temp artifacts — never touches node_modules, vendor, or Docker
- Port conflict detection: automatic check before start, abort on conflict with red warning
- All scripts must be executable (`chmod +x`)

```bash
- [ ] **Step 1: Write scripts/lib/common.sh**

Create the shared library with all functions above. Key implementation details:

- `load_env()`: Search upward from `$PWD` for `.env`, fall back to project root. Source the file. Handle missing `.env` gracefully with a warning.
- `get_port()`: Use `${!var_name}` for indirect variable expansion. Fallback to second argument.
- `check_ports_conflict()`: Extract all PORT_* values, sort and find duplicates. Print conflicts using `log_error()`.
- `write_pid()`: Create `~/.eregen/pids/` directory if needed. Write PID + timestamp.
- `read_pid()`: Read PID file, check if process exists with `kill -0`. Return empty if dead.
- `kill_pid()`: Read PID, send SIGTERM, wait 3s, SIGKILL if still alive. Remove PID file.
- `check_process_running()`: Prefer `lsof -i :$port`, fallback to `ss -tlnp`, then `netstat`.
- `log_*` functions: Use ANSI colors — green for info, yellow for warn, red for error, cyan for success. Fallback to plain text if terminal doesn't support colors.

```bash
#!/usr/bin/env bash
# Eregen Platform - Common Library
# Shared functions for all startup scripts.
# © 2026 Eregen (颐贞). All rights reserved.

set -euo pipefail

# --- Colors ---
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
CYAN='\033[0;36m'
BOLD='\033[1m'
NC='\033[0m' # No Color

log_info()    { echo -e "${GREEN}[INFO]${NC} $*"; }
log_warn()    { echo -e "${YELLOW}[WARN]${NC} $*"; }
log_error()   { echo -e "${RED}[ERROR]${NC} $*"; }
log_success() { echo -e "${CYAN}[OK]${NC} $*"; }
log_header()  { echo -e "\n${BOLD}=== $* ===${NC}\n"; }

# --- Environment ---
PROJECT_ROOT=""
load_env() {
  # Search upward for .env
  local search_dir="$PWD"
  while [ "$search_dir" != "/" ]; do
    if [ -f "$search_dir/.env" ]; then
      PROJECT_ROOT="$search_dir"
      set -a; source "$search_dir/.env"; set +a
      log_info "Loaded .env from $PROJECT_ROOT"
      return 0
    fi
    search_dir="$(dirname "$search_dir")"
  done
  log_warn "No .env found — using hardcoded defaults"
  PROJECT_ROOT="${PWD}"
  return 0
}

get_port() {
  local var_name="PORT_${1^^}"
  local default="${2:-}"
  echo "${!var_name:-$default}"
}

# --- Port Conflict Detection ---
check_ports_conflict() {
  local ports=()
  local names=()

  # Collect all PORT_* variables from environment
  while IFS='=' read -r key value; do
    if [[ "$key" =~ ^PORT_[A-Z_]+$ ]] && [[ "$value" =~ ^[0-9]+$ ]]; then
      ports+=("$value")
      names+=("${key#PORT_}")
    fi
  done < <(compgen -e | grep '^PORT_' || true)

  # Also scan .env directly for completeness
  if [ -n "$PROJECT_ROOT" ] && [ -f "$PROJECT_ROOT/.env" ]; then
    while IFS='=' read -r key value; do
      key=$(echo "$key" | tr -d ' ')
      value=$(echo "$value" | tr -d ' ' | tr -d '"')
      if [[ "$key" =~ ^PORT_[A-Z_]+$ ]] && [[ "$value" =~ ^[0-9]+$ ]]; then
        # Check if already seen
        local found=false
        for p in "${ports[@]:-}"; do
          if [ "$p" = "$value" ]; then found=true; break; fi
        done
        if [ "$found" = false ]; then
          ports+=("$value")
          names+=("${key}")
        fi
      fi
    done < "$PROJECT_ROOT/.env"
  fi

  # Find duplicates
  local dupes
  dupes=$(printf '%s\n' "${ports[@]}" | sort | uniq -d)
  if [ -n "$dupes" ]; then
    log_error "Port conflicts detected!"
    while IFS= read -r port; do
      local svc_names=()
      for i in "${!ports[@]}"; do
        [ "${ports[$i]}" = "$port" ] && svc_names+=("${names[$i]}")
      done
      log_error "  Port $port used by: ${svc_names[*]}"
    done <<< "$dupes"
    return 1
  fi
  log_success "No port conflicts detected"
  return 0
}

# --- PID Management ---
PID_DIR="$HOME/.eregen/pids"

ensure_pid_dir() {
  mkdir -p "$PID_DIR"
}

write_pid() {
  local service="$1"
  local pid="$2"
  ensure_pid_dir
  echo "${pid}" > "$PID_DIR/${service}.pid"
  echo "$(date +%s)" > "$PID_DIR/${service}.timestamp"
}

read_pid() {
  local service="$1"
  local pid_file="$PID_DIR/${service}.pid"
  [ -f "$pid_file" ] || return 1
  local pid
  pid=$(cat "$pid_file")
  # Check if process is alive
  if kill -0 "$pid" 2>/dev/null; then
    echo "$pid"
    return 0
  else
    # Stale PID file
    rm -f "$pid_file" "$PID_DIR/${service}.timestamp"
    return 1
  fi
}

kill_pid() {
  local service="$1"
  local pid_file="$PID_DIR/${service}.pid"
  if [ ! -f "$pid_file" ]; then
    return 0
  fi
  local pid
  pid=$(cat "$pid_file")
  if kill -0 "$pid" 2>/dev/null; then
    log_info "Stopping $service (PID $pid)..."
    kill "$pid" 2>/dev/null || true
    # Wait up to 5 seconds
    local waited=0
    while [ $waited -lt 5 ]; do
      if ! kill -0 "$pid" 2>/dev/null; then break; fi
      sleep 1
      waited=$((waited + 1))
    done
    # Force kill if still alive
    if kill -0 "$pid" 2>/dev/null; then
      kill -9 "$pid" 2>/dev/null || true
    fi
    log_info "$service stopped"
  fi
  rm -f "$pid_file" "$PID_DIR/${service}.timestamp"
}

check_process_running() {
  local port="$1"
  if command -v lsof &>/dev/null; then
    lsof -i :"$port" &>/dev/null && return 0
  elif command -v ss &>/dev/null; then
    ss -tlnp &>/dev/null | grep -q ":$port " && return 0
  elif command -v netstat &>/dev/null; then
    netstat -tlnp &>/dev/null | grep -q ":$port " && return 0
  fi
  return 1
}

# --- Service Discovery ---
list_available_services() {
  log_header "Available Services"
  echo "  Cloud:    api-server push-service data-pipeline admin-api gateway"
  echo "  B2B:      hospital-api community-platform insurance-integration"
  echo "  Apps:     family-app admin-web website miniprogram"
  echo "  Firmware: bracelet pillbox medical-wristband"
  echo ""
  echo "  Groups:   cloud b2b apps firmware all"
}

#!/usr/bin/env bash
# Eregen Platform - Go Service Library
# © 2026 Eregen (颐贞). All rights reserved.

source "$(dirname "$0")/common.sh"

# Go service configuration
# Format: SERVICE_NAME:DIR:PORT_VAR:DEFAULT_PORT
declare -A GO_SERVICES=(
  ["api-server"]="cloud/api-server:PORT_API_SERVER:8080"
  ["push-service"]="cloud/push-service:PORT_PUSH_SERVICE:8085"
  ["data-pipeline"]="cloud/data-pipeline:PORT_DATA_PIPELINE:8087"
  ["admin-api"]="cloud/admin-api:PORT_ADMIN_API:8089"
  ["hospital-api"]="b2b/hospital-api:PORT_HOSPITAL_API:8082"
  ["community-platform"]="b2b/community-platform:PORT_COMMUNITY_PLATFORM:8083"
  ["insurance-integration"]="b2b/insurance-integration:PORT_INSURANCE_INTEGRATION:8084"
  ["gateway"]="cloud/gateway::"
)

go_start() {
  local service="$1"
  local extra_port="${2:-}"

  # Validate service name
  if [ -z "${GO_SERVICES[$service]+x}" ]; then
    log_error "Unknown Go service: $service"
    list_available_services
    return 1
  fi

  local config="${GO_SERVICES[$service]}"
  local dir port_var default_port
  IFS=':' read -r dir port_var default_port <<< "$config"

  local full_dir="$PROJECT_ROOT/$dir"
  if [ ! -d "$full_dir" ]; then
    log_error "Directory not found: $full_dir"
    return 1
  fi

  # Check if already running
  if read_pid "$service" >/dev/null 2>&1; then
    log_warn "$service is already running (PID $(read_pid "$service"))"
    return 0
  fi

  # Determine port
  local port
  if [ -n "$extra_port" ]; then
    port="$extra_port"
  elif [ -n "$port_var" ]; then
    port=$(get_port "$port_var")
  else
    port="$default_port"
  fi

  # Build environment
  local env_args="-e PORT=$port"
  case "$service" in
    api-server)
      env_args+=" -e JWT_SECRET=${JWT_SECRET:-dev-secret} -e DB_URL=${DB_URL:-postgres://eregen:eregen@localhost:5432/eregen?sslmode=disable}"
      ;;
    admin-api)
      env_args+=" -e JWT_SECRET=${JWT_SECRET:-admin-secret} -e SQLITE_PATH=${SQLITE_PATH:-eregen-admin.db}"
      ;;
    push-service)
      env_args+=" -e PUSH_SERVICE_PORT=$port"
      ;;
    data-pipeline)
      env_args+=" -e PIPELINE_PORT=$port"
      ;;
    hospital-api|community-platform|insurance-integration)
      env_args+=" -e DATABASE_URL=${DATABASE_URL:-postgres://postgres:password@localhost:5432/eregen_b2b?sslmode=disable}"
      ;;
  esac

  log_info "Starting $service on port $port ..."
  cd "$full_dir"
  # Run in background, capture PID
  env $env_args go run ./cmd > "$PID_DIR/${service}.log" 2>&1 &
  local pid=$!
  write_pid "$service" "$pid"

  # Wait for server to be ready
  if [ -n "$port" ]; then
    local retries=30
    while [ $retries -gt 0 ]; do
      if check_process_running "$port" 2>/dev/null; then
        log_success "$service started on port $port (PID $pid)"
        log_info "Logs: tail -f $PID_DIR/${service}.log"
        return 0
      fi
      sleep 1
      retries=$((retries - 1))
    done
    log_warn "$service may still be starting — check logs: $PID_DIR/${service}.log"
  else
    # MQTT service (gateway) — no port to check
    sleep 2
    log_success "$service started (PID $pid, no HTTP port)"
    log_info "Logs: tail -f $PID_DIR/${service}.log"
  fi
}

go_stop() {
  local service="$1"
  if [ -z "${GO_SERVICES[$service]+x}" ]; then
    log_error "Unknown Go service: $service"
    return 1
  fi
  kill_pid "$service"
}

go_status() {
  local service="$1"
  if [ -z "${GO_SERVICES[$service]+x}" ]; then
    log_error "Unknown Go service: $service"
    return 1
  fi
  local pid
  if pid=$(read_pid "$service" 2>/dev/null); then
    local port
    if [ "$service" = "gateway" ]; then
      port="(MQTT)"
    else
      local config="${GO_SERVICES[$service]}"
      local port_var
      IFS=':' read -r _ port_var _ <<< "$config"
      port=$(get_port "$port_var")
    fi
    log_success "$service is running (PID $pid, port $port)"
  else
    log_error "$service is not running"
  fi
}

hugo_start() {
  local site="$1"
  local extra_port="${2:-}"

  if [ -z "${HUGO_SITES[$site]+x}" ]; then
    log_error "Unknown Hugo site: $site"
    log_info "Available: ${!HUGO_SITES[@]}"
    return 1
  fi

  local full_dir="$PROJECT_ROOT/${HUGO_SITES[$site]}"
  if [ ! -d "$full_dir" ]; then
    log_error "Directory not found: $full_dir"
    return 1
  fi

  if read_pid "$site" >/dev/null 2>&1; then
    log_warn "$site is already running (PID $(read_pid "$site"))"
    return 0
  fi

  if ! command -v hugo &>/dev/null; then
    log_error "Hugo not found. Install from https://gohugo.io"
    return 1
  fi

  local port
  if [ -n "$extra_port" ]; then
    port="$extra_port"
  else
    port=$(get_port "WEBSITE")
  fi

  log_info "Starting $site on port $port..."
  cd "$full_dir"

  hugo server --bind 0.0.0.0 --port "$port" > "$PID_DIR/${site}.log" 2>&1 &
  local pid=$!
  write_pid "$site" "$pid"

  local retries=30
  while [ $retries -gt 0 ]; do
    if check_process_running "$port" 2>/dev/null; then
      log_success "$site started on port $port (PID $pid)"
      log_info "URL: http://localhost:$port"
      log_info "Logs: tail -f $PID_DIR/${site}.log"
      return 0
    fi
    sleep 1
    retries=$((retries - 1))
  done
  log_warn "$site may still be starting — check logs: $PID_DIR/${site}.log"
}

hugo_stop() {
  local site="$1"
  if [ -z "${HUGO_SITES[$site]+x}" ]; then
    log_error "Unknown Hugo site: $site"
    return 1
  fi
  kill_pid "$site"
}

hugo_status() {
  local site="$1"
  if [ -z "${HUGO_SITES[$site]+x}" ]; then
    log_error "Unknown Hugo site: $site"
    return 1
  fi
  local pid
  if pid=$(read_pid "$site" 2>/dev/null); then
    log_success "$site is running (PID $pid)"
  else
    log_error "$site is not running"
  fi
}

firmware_build() {
  local target="$1"
  local variant="${2:-}"  # basic, smart, auto for pillbox; plus, pro for bracelet

  if [ -z "${FIRMWARE_TARGETS[$target]+x}" ]; then
    log_error "Unknown firmware target: $target"
    log_info "Available: ${!FIRMWARE_TARGETS[@]}"
    return 1
  fi

  if [ ! -d "$full_dir" ]; then
    log_error "Directory not found: $full_dir"
    return 1
  fi

  log_info "Building firmware: $target${variant:+/$variant}..."

    bracelet)
      if ! command -v arm-none-eabi-gcc &>/dev/null; then
        log_error "Arm GNU Toolchain not found"
        log_info "Install: brew install arm-none-eabi-gcc"

        cd "$variant"

      cmake --build build -j$(sysctl -n hw.ncpu 2>/dev/null || nproc)
      ;;
    pillbox|medical-wristband)

      if [ "$target" = "pillbox" ] && [ -n "$variant" ]; then
        cd "$variant"
      fi
      idf.py build
      ;;
  esac

  log_success "$target built successfully"
  log_info "Flash: cd $full_dir && idf.py flash (ESP32) or openocd + arm-none-eabi-size (GD32)"

  case "$group" in

      docker compose -f docker-compose.yml up -d
      ;;
    cloud)
      # Start infra + cloud services

      docker compose -f docker-compose.yml up -d gateway api-server push-service data-pipeline admin-api

      log_info "Groups: all, cloud, b2b, monitoring, frontend"
      return 1
      ;;
  esac

  log_success "Docker services started"
  log_info "Check status: docker compose -f $compose_file ps"
